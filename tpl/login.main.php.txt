<?php

use TSF\Gateway\TokenAuth;
use TSF\Http\App;
use TSF\Http\Request;
use TSF\Http\Response;
use TSF\Kernel\Bootstrap;
use TSF\Kernel\Config;
use TSF\Kernel\Logger;

/**
 * 登录服：Swoole HttpServer + AOT + 路由。
 *
 * ## 加接口只要一行
 *
 * ```php
 * $app->post("/notice", NoticeHandler::class);      // 推荐：类名，可单测
 * $app->get("/ping", fn (Request $r): Response => Response::ok());  // 临时接口
 * ```
 *
 * 路径参数用 `:name`：`$app->get("/player/:uid", ...)`，handler 里 `$req->param("uid")`。
 *
 * ## 统一响应
 *
 * `Response::ok($data)`   → `{"code":0,"data":...}`
 * `Response::fail(1001, "name required", 400)` → `{"code":1001,"msg":"...","status":400}`
 *
 * **HTTP 状态码表达传输层结果，code 表达业务结果** —— 两个都要，别只用一个。
 *
 * ## token
 *
 * `TokenAuth::sign($uid, $ttl)` 签，网关侧 `resolve()` 验，**是同一个类**，
 * 所以不会出现两边算法对不上。网关配置里填同一个 secret 即可。
 */
function main(): void
{
    $base = __DIR__;
    Bootstrap::boot($base, "app");
    Bootstrap::ensureDirs([$base . "/var/log"]);

    $secret = Config::getString("login.secret", "");
    $ttl = Config::getInt("login.token_ttl", 300);
    $gateway = Config::getString("login.gateway", "");
    $auth = new TokenAuth([], $secret);

    $app = new App(
        Config::getString("login.host", "0.0.0.0"),
        Config::getInt("login.port", 9101),
    );

    // ---- 存活检查 ----
    $app->get("/health", function (Request $req): Response {
        return Response::ok(["ok" => true]);
    });

    // ---- 登录 ----
    $app->post("/login", function (Request $req) use ($auth, $ttl, $gateway): Response {
        $name = $req->postOf("name");
        if ($name === "") {
            // 也支持 JSON body
            $body = $req->json();
            $name = (string) ($body["name"] ?? "");
        }
        if ($name === "") {
            return Response::fail(1001, "name required", 400);
        }

        // 真实项目这里查库拿自增 id；示例用稳定 hash 派生
        $h = crc32($name);
        if ($h < 0) {
            $h += 4294967296;
        }
        $uid = (int) ($h % 900000000) + 100000000;

        $token = $auth->sign($uid, $ttl);
        Logger::info("login", ["uid" => $uid, "name" => $name]);

        return Response::ok([
            "uid" => $uid,
            "token" => $token,
            "gateway" => $gateway,
            "ttl" => $ttl,
        ]);
    });

    $app->run();
}
