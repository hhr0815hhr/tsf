<?php

namespace Game\Domain;

/**
 * 给脚本看的只读视图。
 *
 * 只暴露方法，不暴露属性：脚本能读状态、能问"A 成不成"，但写不了。
 * 实测方法调用约 45 ns，白名单字段访问约 410 ns——视图不只是更安全，也更快。
 *
 * 加字段的方式是加方法，不是加 public 属性。加属性等于把写权限也一起给出去了。
 */
final class PlayerView
{
    private int $uid;

    private int $level;

    private int $gold;

    private int $vip;

    public function __construct(int $uid, int $level, int $gold, int $vip)
    {
        $this->uid = $uid;
        $this->level = $level;
        $this->gold = $gold;
        $this->vip = $vip;
    }

    public function uid(): int
    {
        return $this->uid;
    }

    public function level(): int
    {
        return $this->level;
    }

    public function gold(): int
    {
        return $this->gold;
    }

    public function vip(): int
    {
        return $this->vip;
    }

    /** 能力方法：判定留在 AOT，脚本只决定"什么时候用" */
    public function isVip(): bool
    {
        return $this->vip > 0;
    }
}
