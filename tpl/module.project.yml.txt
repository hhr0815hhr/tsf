# wasm 热更模块工程。
#
# 编译：tsf module build .        （在模块目录里跑，或 tsf module build modules/{{name}}）
# 产物：dist/{{name}}.cwasm
#
# ⚠️ 下面两个值由脚手架从 pkg/wasm/src/Iface.php 填入，**不要手改**。
#    它们必须和 typephp 运行时 tp_open() 里硬编码的 world 一致，
#    不一致时加载返回 -1 且不告诉你原因。
wasm-package: {{wasm_package}}
wasm-world: {{wasm_world}}

name: {{name}}
wasm: component
# library：只导出函数，没有自己的 main（绝大多数业务模块都是这个）
wasm-mode: library

# phpx 的 wasm 头用了 std::abort / std::fill 却没 include，
# clang 22+ 不再间接包含（旧版 libc++ 恰好包着）。这两行不能删。
cxx-flags: |
  -include cstdlib
  -include algorithm

sources:
  - lib.php
